# ============================================================
# SMOKE 测试项配置
# ------------------------------------------------------------
# 每个测试项就是一个 [块]，一个块内包含它的开关和全部参数。
#
#   enable = yes   →  执行该项
#   enable = no    →  跳过该项
#
# 参数含义：
#   FIO_CYCLES      电源循环次数
#   IGNORE_ERROR    MachineCheck 不一致时是否继续 (yes/no)
#   FIO_DISKS       指定数据盘，如 sdb,sdc；留空 = 全部数据盘
#   STRESS_MONITOR  是否开启后台压力监控 (yes/no)
#   MONITOR_RUNTIME 后台监控时长（秒），留空 = 默认
#
# 实际执行顺序固定为：reboot → dc → lawdisk → filesystem → mix

# ============================================================


# ==================== lawdisk =====================
[lawdisk]
enable          = yes
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = yes
MONITOR_RUNTIME = 300


# ==================== filesystem ==================
[filesystem]
enable          = yes
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = yes
MONITOR_RUNTIME = 300


# ==================== mix =========================
[mix]
enable          = no
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = no
MONITOR_RUNTIME =


# ==================== reboot ======================
[reboot]
enable          = no
FIO_CYCLES      = 10
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = no
MONITOR_RUNTIME =


# ==================== dc ==========================
[dc]
enable          = no
FIO_CYCLES      = 10
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = no
MONITOR_RUNTIME =
