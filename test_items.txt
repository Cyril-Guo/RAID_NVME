# ============================================================
# SMOKE 测试项配置
# ------------------------------------------------------------
# 每个测试项就是一个 [块]，一个块内包含它的开关和全部参数。
#
#   enable = yes   →  执行该项
#   enable = no    →  跳过该项
#
# 想跑哪个，就把它块里的 enable 改成 yes；参数按需改，不改用默认值。
#
# 参数含义：
#   FIO_CYCLES      电源循环次数（仅 reboot / dc 有）
#   IGNORE_ERROR    MachineCheck 不一致时是否继续 (yes/no)
#   FIO_DISKS       指定数据盘，如 sdb,sdc；留空 = 全部数据盘
#   STRESS_MONITOR  是否开启后台压力监控 (yes/no)
#   MONITOR_RUNTIME 后台监控时长（秒），留空 = 默认
#
# 实际执行顺序固定为：reboot → dc → lawdisk → filesystem → mix
#
# 注：停止/清理(restore)已移到 Jenkins Web 界面的 RESTORE 选项，随时可触发，
#     不再作为测试项在此配置。
# ============================================================


# ==================== 裸盘压力测试 ====================
[lawdisk]
enable          = yes
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = yes
MONITOR_RUNTIME = 300


# ==================== 文件系统压力测试 ====================
[filesystem]
enable          = no
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = no
MONITOR_RUNTIME =


# ==================== 混合 IO 压力测试 ====================
[mix]
enable          = no
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = no
MONITOR_RUNTIME =


# ==================== Reboot 电源循环 ====================
[reboot]
enable          = no
FIO_CYCLES      = 10
IGNORE_ERROR    = no
FIO_DISKS       =
STRESS_MONITOR  = no
MONITOR_RUNTIME =


# ==================== DC 掉电循环 ====================
[dc]
enable          = no
FIO_CYCLES      = 10

FIO_DISKS       =
